export async function createOneTimeProduct(sku, optionId, cents) {
  const pkg = packageName();
  const price = {
    currencyCode: "INR",
    units: String(Math.floor(cents / 100)),
    nanos: cents % 100 * 10_000_000
  };
  const converted = await googleRequest("POST", `/applications/${encodeURIComponent(pkg)}/pricing:convertRegionPrices`, {
    price
  }, "catalog.price-conversion");
  if (!converted.regionVersion?.version || !converted.convertedRegionPrices?.IN) throw new Error("Google Play did not return an India regions version");
  // Only India is enabled. The Admin catalog is INR-only; other regions require
  // a separate currency and tax design, rather than silent conversion.
  const body = {
    packageName: pkg,
    productId: sku,
    listings: [
      {
        languageCode: "en-US",
        title: `MedVeda content INR ${cents / 100}`,
        description: "One purchase of MedVeda digital content selected in the app."
      }
    ],
    purchaseOptions: [
      {
        purchaseOptionId: optionId,
        buyOption: {
          legacyCompatible: true,
          multiQuantityEnabled: false
        },
        regionalPricingAndAvailabilityConfigs: [
          {
            regionCode: "IN",
            price,
            availability: "AVAILABLE"
          }
        ]
      }
    ]
  };
  const query = new URLSearchParams({
    updateMask: "listings,purchaseOptions",
    allowMissing: "true",
    "regionsVersion.version": converted.regionVersion.version
  });
  await googleRequest("PATCH", `/applications/${encodeURIComponent(pkg)}/onetimeproducts/${encodeURIComponent(sku)}?${query}`, body, "catalog.patch");
}
