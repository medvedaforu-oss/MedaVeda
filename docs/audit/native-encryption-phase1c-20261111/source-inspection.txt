83: /**
84:  * Applies the necessary configuration for publishing to the local Maven repository.
85:  * It need to be called when DSL is finalized.
86:  */
87: internal fun Project.applyPublishing(expoModulesExtension: ExpoModuleExtension) {
88:   if (!expoModulesExtension.canBePublished) {
89:     createEmptyExpoPublishTask()
90:     createEmptyExpoPublishToMavenLocalTask()
91:     return
92:   }
93: 
94:   val libraryExtension = androidLibraryExtension()
95: 
96:   libraryExtension
97:     .applyPublishingVariant()
98: 
99:   afterEvaluate {
100:     val publicationInfo = PublicationInfo(this)
101: 
102:     publishingExtension()
103:       .publications
104:       .createReleasePublication(
105:         publicationInfo,
106:         expoModulesExtension.pomConfigurator
107:       )
108: 
109:     createExpoPublishToMavenLocalTask(publicationInfo, expoModulesExtension)
110: 
111:     val npmLocalRepositoryRelativePath = "local-maven-repo"
112:     val npmLocalRepository = File("${project.projectDir.parentFile}/${npmLocalRepositoryRelativePath}").toURI()
113:     publishingExtension().repositories.mavenLocal { mavenRepo ->
114:       mavenRepo.name = "NPMPackage"
115:       mavenRepo.url = npmLocalRepository
116:     }
117: 
118:     createExpoPublishTask(publicationInfo, expoModulesExtension, npmLocalRepositoryRelativePath)
119:   }
120: }
